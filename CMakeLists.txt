cmake_minimum_required(VERSION 3.24)
project(VolturaBooks VERSION 0.1.1 LANGUAGES C CXX RC)
configure_file(src/version.h.in generated/books_version.h @ONLY)
include_directories(${CMAKE_CURRENT_BINARY_DIR}/generated)
if(NOT WIN32)
  message(FATAL_ERROR "Voltura Books requires Windows")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
set(BUILD_CURL_EXE OFF CACHE BOOL "" FORCE)
set(BUILD_LIBCURL_DOCS OFF CACHE BOOL "" FORCE)
set(BUILD_MISC_DOCS OFF CACHE BOOL "" FORCE)
set(ENABLE_CURL_MANUAL OFF CACHE BOOL "" FORCE)
set(CURL_USE_LIBSSH2 OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(CURL_USE_SCHANNEL ON CACHE BOOL "" FORCE)
set(CURL_USE_OPENSSL OFF CACHE BOOL "" FORCE)
set(CURL_STATIC_CRT ON CACHE BOOL "" FORCE)
set(CURL_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
set(CURL_DISABLE_TESTS ON CACHE BOOL "" FORCE)
set(HTTP_ONLY OFF CACHE BOOL "" FORCE)
foreach(protocol DICT FILE FTP GOPHER HTTP IMAP LDAP LDAPS MQTT POP3 RTSP SMB TELNET TFTP WS)
  set(CURL_DISABLE_${protocol} ON CACHE BOOL "" FORCE)
endforeach()
set(CURL_ZLIB OFF CACHE BOOL "" FORCE)
set(CURL_BROTLI OFF CACHE BOOL "" FORCE)
set(CURL_ZSTD OFF CACHE BOOL "" FORCE)
set(USE_LIBIDN2 OFF CACHE BOOL "" FORCE)
set(USE_NGHTTP2 OFF CACHE BOOL "" FORCE)
set(CURL_USE_LIBPSL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(curl URL https://curl.se/download/curl-8.22.0.tar.xz
  URL_HASH SHA256=f7ef3ae8a22e521f289803fe93543eb64c329b58aa73a9e224dfd915a2a5f4f7)
FetchContent_MakeAvailable(curl)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(INSTALL_PROJECT OFF CACHE BOOL "" FORCE)
FetchContent_Declare(miniz URL https://github.com/richgel999/miniz/archive/refs/tags/3.1.0.tar.gz
  URL_HASH SHA256=09569fc19d060ac9f5999ba9356728c2494ebe6a24ac0eb0a6b6ae3d396cfea6)
FetchContent_MakeAvailable(miniz)
add_library(books_cover src/cover.cpp)
target_include_directories(books_cover PUBLIC src)
target_compile_definitions(books_cover PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(books_cover PUBLIC miniz windowscodecs xmllite shlwapi ole32 gdi32 windowsapp shcore propsys)
add_library(books_core src/core.cpp src/settings.cpp src/history.cpp src/mail_setup.cpp src/drop.cpp src/shell_selection.cpp)
target_include_directories(books_core PUBLIC src)
target_compile_definitions(books_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(books_core PUBLIC CURL::libcurl advapi32 shell32 ole32 bcrypt winhttp dnsapi ws2_32 normaliz xmllite shlwapi)
add_library(books_theme src/theme.cpp src/placeholder.cpp src/book_information.cpp src/list_scrollbar.cpp src/file_icons.cpp src/about.cpp src/update.cpp)
target_include_directories(books_theme PUBLIC src)
target_compile_definitions(books_theme PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(books_theme PUBLIC comctl32 dwmapi uxtheme advapi32 msimg32 shell32 winhttp crypt32 bcrypt windowsapp)
add_executable(VolturaBooks WIN32 src/main.cpp src/integration.cpp src/browser.cpp src/app.rc)
target_link_libraries(VolturaBooks PRIVATE books_core books_cover books_theme comctl32 comdlg32 msimg32)
target_link_options(VolturaBooks PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest)
foreach(target books_core VolturaBooks)
  target_compile_options(${target} PRIVATE /W4 /utf-8 /permissive-)
endforeach()
